            def call_marin_api(endpoint, payload):

                r = requests.post(f"{MARIN_API_URL}{endpoint}", json=payload, headers=headers)
                r.raise_for_status()
                return r.json()

            def run_tool(tool_name, tool_input):
                """Execute a tool via Marin's command API or direct subprocess."""
                if tool_name == "TERMINAL":
                    # Use the existing /command endpoint
                    return call_marin_api("/command", {"cmd": tool_input, "timeout": 60})
                elif tool_name == "WEB_SEARCH":
                    # Example: use curl or a Python script; here we call a helper script
                    cmd = f"python3 -c \"import requests, sys; r=requests.get('https://api.duckduckgo.com/?q={requests.utils.quote(tool_input)}&format=json'); print(r.json().get('Abstract','No abstract'))\""

                    return {"output": result.stdout.strip()} if result.returncode == 0 else {"error": result.stderr}
                elif tool_name == "READ_FILE":
                    path = Path(tool_input).expanduser().resolve()
                    return {"output": path.read_text()[:2000]}  # truncate

                    path, content = tool_input.split("::", 1)
                    Path(path).write_text(content)

    Do not emit JSON, code blocks, or any other structured data—only natural language and the TOOL line when needed.
